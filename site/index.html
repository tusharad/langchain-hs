{-# LANGUAGE OverloadedStrings #-}
import Langchain.Prelude

-- Compose pure AST pipelines with (|>>) and (&>&)
ragPipeline :: RunnableTree IO Text Text
ragPipeline =
      runLambda (\q -> (q, q))                      -- duplicate query
  |>> (fetchDocuments &>& generateFollowup)          -- parallel branch
  |>> runLambda (\(docs, fup) -> renderPrompt docs fup)
  |>> invokeLLM model
  |>> parseStructuredOutput

-- Zero IO during construction. Full AST introspection & optimization before interpretation!
main :: IO ()
main = do
  output <- interpret ragPipeline "Explain Monads in 3 sentences"
  print output

data AgentState = AgentState { messages :: [Message], currentPhase :: Text }

-- Construct cyclic workflow with STM TVar checkpointing and SQLite persistence
workflow :: StateGraph AgentState IO
workflow = emptyStateGraph appendMessagesReducer
  & addNode "planner" plannerNode
  & addNode "executor" executorNode
  & addNode "reviewer" reviewerNode
  & addEdge startNodeId "planner"
  & addEdge "planner" "executor"
  & addConditionalEdge "executor" shouldReview 
      [ ("continue", "reviewer"), ("done", endNodeId) ]
  & addEdge "reviewer" "planner" -- cyclic feedback loop!

compiled = compileGraph workflow (Just memoryCheckpointer)

  -- Connect to any MCP server via stdio or HTTP JSON-RPC 2.0

  -- Introspect server tools and automatically convert to native Tool definitions
  mcpTools <- listMcpTools client
  let nativeTools = map mcpToolToLangchainTool mcpTools
  
  -- Bind tools to your ReAct or Plan-and-Execute Agent
  let agent = createReActAgent model nativeTools defaultAgentConfig
  res <- runReActAgent agent "Inspect the current filesystem and count .hs files"
  print res
